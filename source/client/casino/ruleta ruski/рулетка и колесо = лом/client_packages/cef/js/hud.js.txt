data: {
fortuneActive: true,
        fortuneToggle: true,
        fortuneMin: 0,
        fortuneHour: 0,
},
methods: {


 showFortune(hour, minute) {
            this.fortuneActive = true;
            this.fortuneHour =  hour;

        toggleFortune(status) {
            this.fortuneToggle = status;
        },

}
